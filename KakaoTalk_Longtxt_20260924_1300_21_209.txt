function inSel(p){
  const v=sel.q;let c=false;
  for(let i=0,j=3;i<4;j=i++){if((v[i].y>p.y)!==(v[j].y>p.y)&&p.x<(v[j].x-v[i].x)*(p.y-v[i].y)/(v[j].y-v[i].y)+v[i].x)c=!c}
  return c;
}
function hitHandle(p){let b=-1,bd=18;sel.q.forEach((c,i)=>{const d=Math.hypot(p.x-c.x,p.y-c.y)*view.s;if(d<=bd){bd=d;b=i}});return b}
function pathOf(c,pts){c.beginPath();pts.forEach((q,i)=>i?c.lineTo(q.x,q.y):c.moveTo(q.x,q.y));c.closePath()}
function convex(q){let s=0;for(let i=0;i<4;i++){const a=q[i],b=q[(i+1)%4],c=q[(i+2)%4],z=(b.x-a.x)*(c.y-b.y)-(b.y-a.y)*(c.x-b.x);if(Math.abs(z)<1e-6)return false;const g=z>0?1:-1;if(s&&g!==s)return false;s=g}return true}
/* 원근 변형: 단위 정사각형 -> 사각형 호모그래피 */
function sq2quad(q){
  const x0=q[0].x,y0=q[0].y,x1=q[1].x,y1=q[1].y,x2=q[2].x,y2=q[2].y,x3=q[3].x,y3=q[3].y;
  const dx1=x1-x2,dx2=x3-x2,dx3=x0-x1+x2-x3,dy1=y1-y2,dy2=y3-y2,dy3=y0-y1+y2-y3,den=dx1*dy2-dx2*dy1;
  const g=(dx3*dy2-dx2*dy3)/den,h=(dx1*dy3-dx3*dy1)/den;
  return{a:x1-x0+g*x1,b:x3-x0+h*x3,c:x0,d:y1-y0+g*y1,e:y3-y0+h*y3,f:y0,g,h};
}
function triDraw(f,S,s0,s1,s2,d0,d1,d2){
  const det=(s1.x-s0.x)*(s2.y-s0.y)-(s2.x-s0.x)*(s1.y-s0.y);if(!det)return;
  const a=((d1.x-d0.x)*(s2.y-s0.y)-(d2.x-d0.x)*(s1.y-s0.y))/det,c=((d2.x-d0.x)*(s1.x-s0.x)-(d1.x-d0.x)*(s2.x-s0.x))/det,
        b=((d1.y-d0.y)*(s2.y-s0.y)-(d2.y-d0.y)*(s1.y-s0.y))/det,d=((d2.y-d0.y)*(s1.x-s0.x)-(d1.y-d0.y)*(s2.x-s0.x))/det,
        e=d0.x-a*s0.x-c*s0.y,g=d0.y-b*s0.x-d*s0.y,
        cx=(d0.x+d1.x+d2.x)/3,cy=(d0.y+d1.y+d2.y)/3,
        grow=v=>{const dx=v.x-cx,dy=v.y-cy,l=Math.hypot(dx,dy)||1;return{x:v.x+dx/l*.8,y:v.y+dy/l*.8}};
  f.save();f.beginPath();[d0,d1,d2].map(grow).forEach((v,i)=>i?f.lineTo(v.x,v.y):f.moveTo(v.x,v.y));f.closePath();f.clip();
  f.setTransform(a,b,c,d,e,g);f.drawImage(S,0,0);f.restore();
}
function renderSel(fast){
  const f=fc.getContext('2d'),S=sel.src,q=sel.q,w=S.width,h=S.height;
  f.setTransform(1,0,0,1,0,0);f.clearRect(0,0,W,H);f.imageSmoothingQuality='high';
  if(Math.abs(q[0].x+q[2].x-q[1].x-q[3].x)<.01&&Math.abs(q[0].y+q[2].y-q[1].y-q[3].y)<.01){
    f.setTransform((q[1].x-q[0].x)/w,(q[1].y-q[0].y)/w,(q[3].x-q[0].x)/h,(q[3].y-q[0].y)/h,q[0].x,q[0].y);
    f.drawImage(S,0,0);f.setTransform(1,0,0,1,0,0);return;
  }
  const M=sq2quad(q),N=fast?8:16,mp=(u,v)=>{const z=M.g*u+M.h*v+1;return{x:(M.a*u+M.b*v+M.c)/z,y:(M.d*u+M.e*v+M.f)/z}};
  for(let i=0;i<N;i++)for(let j=0;j<N;j++){
    const u0=i/N,u1=(i+1)/N,v0=j/N,v1=(j+1)/N,
      s00={x:u0*w,y:v0*h},s10={x:u1*w,y:v0*h},s11={x:u1*w,y:v1*h},s01={x:u0*w,y:v1*h},
      d00=mp(u0,v0),d10=mp(u1,v0),d11=mp(u1,v1),d01=mp(u0,v1);
    triDraw(f,S,s00,s10,s01,d00,d10,d01);triDraw(f,S,s10,s11,s01,d10,d11,d01);
  }
}
function dragSel(e,p){
  const d=sel.drag,q0=d.q0;let q;
  if(d.t==='m'){const dx=Math.round(p.x-d.sx),dy=Math.round(p.y-d.sy);q=q0.map(c=>({x:c.x+dx,y:c.y+dy}))}
  else if(tool==='lpersp'){q=q0.map((c,i)=>i===d.i?{x:p.x,y:p.y}:{x:c.x,y:c.y})}
  else{
    const a=q0[(d.i+2)%4],c=q0[d.i],vx=c.x-a.x,vy=c.y-a.y;
    if(e.shiftKey&&Math.abs(vx)>1&&Math.abs(vy)>1){const fx=Math.max(.05,(p.x-a.x)/vx),fy=Math.max(.05,(p.y-a.y)/vy);q=q0.map(z=>({x:a.x+(z.x-a.x)*fx,y:a.y+(z.y-a.y)*fy}))}
    else{const f=Math.max(.05,((p.x-a.x)*vx+(p.y-a.y)*vy)/(vx*vx+vy*vy));q=q0.map(z=>({x:a.x+(z.x-a.x)*f,y:a.y+(z.y-a.y)*f}))}
  }
  if(!convex(q))return;
  sel.q=q;renderSel(true);drawOv();
}
function makeSel(L,pts){
  let x0=1e9,y0=1e9,x1=-1e9,y1=-1e9;pts.forEach(q=>{x0=Math.min(x0,q.x);y0=Math.min(y0,q.y);x1=Math.max(x1,q.x);y1=Math.max(y1,q.y)});
  x0=Math.max(0,Math.floor(x0));y0=Math.max(0,Math.floor(y0));x1=Math.min(W,Math.ceil(x1));y1=Math.min(H,Math.ceil(y1));
  const w=x1-x0,h=y1-y0;if(w<2||h<2)return;
  pushUndo(L);
  const src=document.createElement('canvas');src.width=w;src.height=h;const f=src.getContext('2d');
  f.save();f.translate(-x0,-y0);pathOf(f,pts);f.clip();f.drawImage(L.c,0,0);f.restore();
  L.ctx.save();pathOf(L.ctx,pts);L.ctx.clip();L.ctx.clearRect(0,0,W,H);L.ctx.restore();
  sel={L,src,q:[{x:x0,y:y0},{x:x1,y:y0},{x:x1,y:y1},{x:x0,y:y1}]};
  placeSel();renderSel(false);drawOv();
}

}
